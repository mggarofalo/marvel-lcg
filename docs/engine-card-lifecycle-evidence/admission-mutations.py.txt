from pathlib import Path
import json, re, shutil, subprocess

root = Path('/Users/michael/Source/marvel-lcg')
target = Path('/tmp/marvel-lifecycle-mutants')
changed = subprocess.check_output(['git', 'diff', '--name-only', '-z', '3246b4a5261fbcf3f33db01c91a6e58b28631b12'], cwd=root).decode().split('\0')
changed += subprocess.check_output(['git', 'ls-files', '--others', '--exclude-standard', '-z'], cwd=root).decode().split('\0')
for name in changed:
    if not name:
        continue
    source, dest = root / name, target / name
    if source.exists():
        dest.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(source, dest)
        dest.touch()
    elif dest.exists():
        dest.unlink()

project = 'tests/Marvel.Content.Tests/Marvel.Content.Tests.csproj'
test = 'SingularQueriesAfterEngagementAreRejectedBeforeMutation'

def run(name, selection):
    log = Path('/tmp/marvel-lifecycle-' + name + '.log')
    with log.open('w') as out:
        result = subprocess.run(['dotnet', 'test', project, '-c', 'Release', '--filter', selection,
                                 '--logger', 'console;verbosity=normal'], cwd=target, stdout=out, stderr=subprocess.STDOUT)
    return result.returncode, log.read_text(), str(log)

rc, output, log = run('admission-baseline', 'FullyQualifiedName~FacedownMinionAdmissionTests')
print(json.dumps({'baseline': 'admission', 'exit': rc, 'log': log}), flush=True)
if rc:
    raise SystemExit('Unmutated baseline failed')

cases = [
    ('M16-engagement-areas', 'src/Marvel.Cards/Run/AbilityAreaMutationQuery.cs',
     '            DeckType.PlayerDeck, DeckType.DiscardPile, DeckType.EngagedEnemiesArea,\n', '', test),
    ('M17-single-movement-target', 'src/Marvel.Cards/Run/AbilityTargetAdmission.cs',
     '            or "returnOwnedToHand" or "addToHand"\n', '', test),
    ('M18-hand-owner', 'src/Marvel.Cards/Run/AbilityCardMovementAdmission.cs',
     'movement.Instruction != AbilityCardInstruction.ReturnOwnedToHand || card.Owner >= 0', 'true',
     'ReturningToAnOwnersHandRequiresAPhysicalPlayerOwner'),
    ('M19-hand-movement-areas', 'src/Marvel.Cards/Run/AbilityAreaMutationQuery.cs',
     '            or "returnOwnedToHand" or "addToHand" or "returnOwnedToDiscard"\n', '',
     'SingularQueriesAfterHandMovementAreRejectedBeforeMutation'),
    ('M20-hand-projection', 'src/Marvel.Cards/Run/AbilityAreaMovementProjection.cs',
     '            or "returnOwnedToHand" or "addToHand" or "returnOwnedToDiscard"\n', '',
     'SingularQueriesAfterHandMovementAreRejectedBeforeMutation'),
]
results = []
for name, path, before, after, intended_test in cases:
    source = target / path
    original = source.read_text()
    if original.count(before) != 1:
        raise SystemExit(f'{name}: expression not unique')
    try:
        source.write_text(original.replace(before, after))
        rc, output, log = run(name, 'FullyQualifiedName~' + intended_test)
        compiled = not re.search(r': error (CS|MSB|NETSDK)', output)
        caught = bool(re.search(r'Failed Marvel\..*' + intended_test, output))
        result = {'name': name, 'test': intended_test, 'compiled': compiled, 'exit': rc,
                  'intendedBehaviorCaught': caught, 'killed': rc != 0 and compiled and caught, 'log': log}
        results.append(result)
        print(json.dumps(result), flush=True)
    finally:
        source.write_text(original)
Path('/tmp/marvel-lifecycle-admission-mutation-results.json').write_text(json.dumps(results, indent=2) + '\n')
if not all(result['killed'] for result in results):
    raise SystemExit('A mutant needs investigation')
