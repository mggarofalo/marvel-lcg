from . import *

# Example Event

def GetAbilities() -> Sequence['Ability']:

    def example_effect(effect: 'Effect', message: 'Message.WhenPlayerInTurn') -> None:
        this = effect.this.CastTo(Event)                # mark this as an Event
        Unused(this)

        this.HealthUnits(effect.targets, 10, effect)    # heal units
        player = this.GetControlByPlayer()

        player.DrawUp(5, effect)                        # draw up cards
        # add more if you want

    return [
        AbilityFactory.WhenInYourPlayTurn(
            AbilityType.Action,
            example_effect,
        ).SetPlay()
        .SetTarget(Friend)                              # effect target
        .LimitOncePerRound()
    ]
